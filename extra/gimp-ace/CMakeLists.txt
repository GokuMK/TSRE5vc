cmake_minimum_required(VERSION 3.22)
project(GimpAce VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GIMP REQUIRED IMPORTED_TARGET gimpui-3.0>=3.0)
include(../common/ace-codec/AceCodec.cmake)
tsre_add_ace_codec(ace_gimp_codec)
add_library(ace_export STATIC src/Export.cpp)
target_include_directories(ace_export PUBLIC src)
target_link_libraries(ace_export PUBLIC ace_gimp_codec)
add_executable(file-tsre-ace src/Plugin.cpp)
set_target_properties(file-tsre-ace PROPERTIES WIN32_EXECUTABLE TRUE)
target_link_libraries(file-tsre-ace PRIVATE ace_export PkgConfig::GIMP)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(file-tsre-ace PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(ace_export PRIVATE -Wall -Wextra -Wpedantic)
endif()
# Use an explicit user plug-in root to install into an existing GIMP profile.
set(GIMP_PLUGIN_DIR "lib/gimp/3.0/plug-ins" CACHE PATH "GIMP plug-in root directory")
install(TARGETS file-tsre-ace RUNTIME DESTINATION "${GIMP_PLUGIN_DIR}/file-tsre-ace")
include(CTest)
if(BUILD_TESTING)
    add_executable(ace_export_tests tests/ExportTests.cpp)
    target_link_libraries(ace_export_tests PRIVATE ace_export)
    add_test(NAME ace_export COMMAND ace_export_tests)
    # Automated dialog driver exists only in this separate test executable.
    add_executable(file-tsre-ace-dialog-test src/Plugin.cpp)
    target_compile_definitions(file-tsre-ace-dialog-test PRIVATE ACE_GIMP_TEST_DIALOG)
    target_link_libraries(file-tsre-ace-dialog-test PRIVATE ace_export PkgConfig::GIMP)
endif()
