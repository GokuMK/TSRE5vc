cmake_minimum_required(VERSION 3.22)
project(AceThumbnails VERSION 1.0.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MINGW OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "AceThumbnails currently requires 64-bit MinGW on Windows")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
get_filename_component(TSRE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(CODEC_SOURCES "${TSRE_ROOT}/src/tsre/texture/AceDocument.cpp"
                  "${TSRE_ROOT}/src/tsre/texture/DxtCodec.cpp")
include(../common/ace-codec/AceCodec.cmake)
tsre_add_ace_codec(ace_thumbnail_codec)
add_library(AceThumbnails SHARED src/Provider.cpp src/Thumbnail.cpp src/Registration.cpp src/exports.def)
set_target_properties(AceThumbnails PROPERTIES PREFIX "")
target_link_libraries(AceThumbnails PRIVATE ace_thumbnail_codec gdi32 ole32 shell32 advapi32 uuid)
target_include_directories(AceThumbnails PRIVATE src)
foreach(target ace_thumbnail_codec AceThumbnails)
    target_compile_definitions(${target} PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0601)
    target_compile_options(${target} PRIVATE -ffunction-sections -fdata-sections)
endforeach()
target_compile_options(AceThumbnails PRIVATE -Wall -Wextra -Wpedantic)
target_link_options(AceThumbnails PRIVATE -static -static-libgcc -static-libstdc++ -Wl,--gc-sections -Wl,--exclude-all-symbols)
include(CTest)
if(BUILD_TESTING)
    add_executable(ace_thumbnail_tests tests/Tests.cpp)
    target_include_directories(ace_thumbnail_tests PRIVATE src)
    target_compile_definitions(ace_thumbnail_tests PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
    target_link_libraries(ace_thumbnail_tests PRIVATE ace_thumbnail_codec gdi32 ole32 shlwapi shell32 advapi32 uuid psapi)
    target_link_options(ace_thumbnail_tests PRIVATE -static -static-libgcc -static-libstdc++ -municode)
    add_test(NAME thumbnails COMMAND ace_thumbnail_tests "$<TARGET_FILE:AceThumbnails>")
    add_test(NAME dll_dependencies COMMAND "${CMAKE_COMMAND}"
        "-DOBJDUMP=${CMAKE_OBJDUMP}" "-DDLL=$<TARGET_FILE:AceThumbnails>"
        -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/Dependencies.cmake")
    # Optional parity oracle: same corpus generator/decoder compiled against real Qt.
    # This option does not add Qt to the DLL or the normal build.
    option(ACE_THUMBNAILS_QT_PARITY "Build a Qt-backed codec parity oracle" OFF)
    add_executable(ace_codec_corpus tests/Corpus.cpp)
    target_link_libraries(ace_codec_corpus PRIVATE ace_thumbnail_codec)
    target_link_options(ace_codec_corpus PRIVATE -static -static-libgcc -static-libstdc++)
    if(ACE_THUMBNAILS_QT_PARITY)
        find_package(Qt6 REQUIRED COMPONENTS Core)
        add_executable(ace_codec_corpus_qt tests/Corpus.cpp ${CODEC_SOURCES})
        target_include_directories(ace_codec_corpus_qt PRIVATE "${TSRE_ROOT}/src")
        target_compile_definitions(ace_codec_corpus_qt PRIVATE MINIZ_NO_ARCHIVE_APIS MINIZ_NO_STDIO)
        target_link_libraries(ace_codec_corpus_qt PRIVATE Qt6::Core)
        add_test(NAME codec_parity COMMAND "${CMAKE_COMMAND}"
            "-DSTD_EXE=$<TARGET_FILE:ace_codec_corpus>" "-DQT_EXE=$<TARGET_FILE:ace_codec_corpus_qt>"
            "-DOUTPUT_DIR=${CMAKE_CURRENT_BINARY_DIR}/parity" -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/Parity.cmake")
        get_filename_component(compiler_bin "${CMAKE_CXX_COMPILER}" DIRECTORY)
        set_tests_properties(codec_parity PROPERTIES ENVIRONMENT_MODIFICATION
            "PATH=path_list_prepend:$<TARGET_FILE_DIR:Qt6::Core>;PATH=path_list_prepend:${compiler_bin}")
    endif()
endif()
